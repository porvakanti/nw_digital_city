# What a developer needs on top of what gets deployed.
#
# Kept separate on purpose. app/requirements.txt is what goes into the
# container, and a test-only library has no business being in a production
# image. run.py installs this file; the Dockerfile installs the other one.

-r app/requirements.txt

# Starlette's TestClient wants httpx2 and warns on every import without it.
# The warning is real, not noise: starlette will drop httpx support from the
# test client. It only affects tests, which is why it lives here.
httpx2>=2.12

# The product film. Pillow draws the text frames, because the ffmpeg that
# imageio-ffmpeg ships carries no drawtext filter, and reads the frames back
# to find where each recording's usable footage starts. Neither library is
# used by the service, which is why they are here and not in
# app/requirements.txt.
Pillow>=10.0
imageio-ffmpeg>=0.5

# The narrated reel. numpy and scipy synthesise and mix the score, edge-tts
# synthesises the narration. Build-time only, like the film's.
numpy>=1.24
scipy>=1.10
edge-tts>=6.1
